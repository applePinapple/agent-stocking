{UI PARAMETER SID MIN=3 MAX=21 DEFAULT=17; ONLY THIS SID CAN SIGNAL.}
{Only unused dependencies removed. Fills remain next-period-open in evaluator.}
DLY:=PERIOD=5;
SPAN:=DATE>=1210101 AND DATE<=1261023;
R2:=BARSCOUNT(C)>=2;
WRHI:=HHV(H,2);
WRLO:=LLV(L,2);
WRDEN:=WRHI-WRLO;
WRV:=-100*(WRHI-C)/IF(WRDEN>0,WRDEN,1);
WROK:=R2 AND WRDEN>0;
B17:=WROK AND WRV<-90;
S17:=R2 AND (C>REF(H,1) OR (WROK AND WRV>-30));
ENTERLONG:DLY AND SPAN AND SID=17 AND B17 AND NOT(S17);
EXITLONG:DLY AND SPAN AND SID=17 AND S17;
