{UI PARAMETER SID MIN=3 MAX=21 DEFAULT=5; ONLY THIS SID CAN SIGNAL.}
{Only unused dependencies removed. Fills remain next-period-open in evaluator.}
DLY:=PERIOD=5;
SPAN:=DATE>=1210101 AND DATE<=1261023;
R20:=BARSCOUNT(C)>=20;
TYP:=(H+L+C)/3;
CTMA:=MA(TYP,20);
CDEV:=AVEDEV(TYP,20);
CCIV:=(TYP-CTMA)/(0.015*IF(CDEV>0,CDEV,1));
CCIOK:=R20 AND CDEV>0;
B5:=CCIOK AND REF(CCIOK,1) AND CCIV>100 AND REF(CCIV,1)<100;
S5:=CCIOK AND CCIV<100;
ENTERLONG:DLY AND SPAN AND SID=5 AND B5 AND NOT(S5);
EXITLONG:DLY AND SPAN AND SID=5 AND S5;
