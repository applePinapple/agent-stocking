{UI PARAMETER SID MIN=3 MAX=21 DEFAULT=9; ONLY THIS SID CAN SIGNAL.}
{Only unused dependencies removed. Fills remain next-period-open in evaluator.}
DLY:=PERIOD=5;
SPAN:=DATE>=1210101 AND DATE<=1261023;
R200:=BARSCOUNT(C)>=200;
R3:=BARSCOUNT(C)>=3;
MA200:=MA(C,200);
MA5:=MA(C,5);
DELTA:=C-REF(C,1);
RUP:=SMA(MAX(DELTA,0),2,1);
RDN:=SMA(ABS(DELTA),2,1);
RSIV:=IF(RDN>0,100*RUP/IF(RDN>0,RDN,1),50);
RSIOK:=R3 AND RDN>0;
B9:=R200 AND RSIOK AND C>MA200 AND RSIV<=5;
S9:=R200 AND C>MA5;
ENTERLONG:DLY AND SPAN AND SID=9 AND B9 AND NOT(S9);
EXITLONG:DLY AND SPAN AND SID=9 AND S9;
