{S21 LOCAL TTM ADAPTATION. NATIVE COMPILE AND OUTPUT CHECKS PENDING.}
{SID MIN=3 MAX=21 DEFAULT=21. DAILY CLOSE SIGNALS, NEXT OPEN FILLS.}
{ORIGINAL-STYLE KC: SMA OF TYPICAL PRICE +/- 1.5*SMA OF H-L. NOT EMA/ATR KC.}
{FORCAST IS THE SELECTED NATIVE REGRESSION FUNCTION; NO CROSS-PLATFORM IDENTITY CLAIM.}
DLY:=PERIOD=5;
SPAN:=DATE>=1210101 AND DATE<=1261023;
RB:=BARSCOUNT(C)>=40;
RS:=BARSCOUNT(C)>=41;
BBM:=MA(C,20);
BSD:=STDP(C,20);
BBU:=BBM+2*BSD;
BBL:=BBM-2*BSD;
KCM:=MA((H+L+C)/3,20);
KCR:=MA(H-L,20);
KCU:=KCM+1.5*KCR;
KCL:=KCM-1.5*KCR;
SQZ:=BBU<KCU AND BBL>KCL;
DCM:=(HHV(H,20)+LLV(L,20))/2;
MDEL:=C-(DCM+BBM)/2;
MOM:=FORCAST(MDEL,20);
B21:=RB AND REF(SQZ,1) AND NOT(SQZ) AND MOM>0 AND MOM>REF(MOM,1);
S21:=RS AND MOM>0 AND MOM<REF(MOM,1) AND REF(MOM,1)>0 AND REF(MOM,1)<REF(MOM,2);
ENTERLONG:DLY AND SPAN AND SID=21 AND B21 AND NOT(S21);
EXITLONG:DLY AND SPAN AND SID=21 AND S21;
