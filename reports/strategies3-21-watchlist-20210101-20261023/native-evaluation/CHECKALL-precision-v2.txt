{CHECKALL PRECISION V2: 15 READ-ONLY DIAGNOSTIC OUTPUTS. NO ORDERS.}
{SCALE 1048576 = 2^20: DIVIDE X OUTPUTS BY THIS EXACT FACTOR.}
{SCALING PRESERVES AVAILABLE NATIVE PRECISION; IT DOES NOT CREATE NEW MARKET PRECISION.}
{UNSCALED -1 IN R2X/R4X IS INVALID DATA, NEVER AN OVERSOLD SIGNAL.}
{AMT IS UNADJUSTED CNY TURNOVER AS REPORTED BY NATIVE AMOUNT.}
{FQ/BC/CAPSH/TURN3/AMT ARE NOT SCALED. MA/SAR/STDP/FORCAST SEMANTICS UNCHANGED.}
SC:=1048576;
FQ:TQFLAG,NODRAW;
BC:BARSCOUNT(C),NODRAW;
OX:O*SC,NODRAW;
HX:H*SC,NODRAW;
LX:L*SC,NODRAW;
CX:C*SC,NODRAW;
DLTA:=C-REF(C,1);
U2:=SMA(MAX(DLTA,0),2,1);
D2:=SMA(ABS(DLTA),2,1);
V2:=IF(D2>0,100*U2/IF(D2>0,D2,1),50);
OK2:=BC>=3 AND D2>0;
U4:=SMA(MAX(DLTA,0),4,1);
D4:=SMA(ABS(DLTA),4,1);
V4:=IF(D4>0,100*U4/IF(D4>0,D4,1),50);
OK4:=BC>=5 AND D4>0;
R2X:IF(OK2,V2*SC,-1),NODRAW;
R4X:IF(OK4,V4*SC,-1),NODRAW;
M200X:MA(C,200)*SC,NODRAW;
CAPSH:FINANCE(1),NODRAW;
SARX:SAR(4,2,20)*SC,NODRAW;
TURN3:SARTURN(4,2,20),NODRAW;
POPX:STDP(C,20)*SC,NODRAW;
BM21:=MA(C,20);
DC21:=(HHV(H,20)+LLV(L,20))/2;
DL21:=C-(DC21+BM21)/2;
FCX:FORCAST(DL21,20)*SC,NODRAW;
AMT:AMOUNT,NODRAW;
