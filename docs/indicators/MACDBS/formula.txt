DIFF:EMA(CLOSE,SHORT)-EMA(CLOSE,LONG),COLORWHITE;
DEA:EMA(DIFF,M),COLORYELLOW;
MACD:2*(DIFF-DEA),NODRAW;
ZERO:0,COLORGRAY;
STICKLINE(MACD>=0,0,MACD,1,0),COLORRED;
STICKLINE(MACD<0,0,MACD,1,0),COLORGREEN;
BSIG:CROSS(MACD,0),NODRAW;
SSIG:CROSS(0,MACD),NODRAW;
GAP:=MAX(ABS(DIFF),ABS(DEA))*0.12+0.01;
DRAWICON(BSIG,MIN(DIFF,DEA)-GAP,1);
DRAWTEXT(BSIG,MIN(DIFF,DEA)-2*GAP,'买'),COLORRED;
DRAWICON(SSIG,MAX(DIFF,DEA)+GAP,2);
DRAWTEXT(SSIG,MAX(DIFF,DEA)+2*GAP,'卖'),COLORGREEN;
