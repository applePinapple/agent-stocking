{SID PARAMETER: MIN 3 MAX 21 DEFAULT 8; THIS FILE REQUIRES SID=8.}
{PREVIOUS FIVE COMPLETED BARS; CHANNEL +/- NATIVE MINDIFF; LONG ONLY.}
{NOT A DYNAMIC STOP ORDER; BOTH SIDES ON SAME BAR: EXIT WINS.}
DLY:=PERIOD=5;
SPAN:=DATE>=1210101 AND DATE<=1261023;
CHREADY:=BARSCOUNT(C)>=6 AND MINDIFF>0;
CHUP:=REF(HHV(H,5),1)+MINDIFF;
CHDN:=REF(LLV(L,5),1)-MINDIFF;
B8:=CHREADY AND H>=CHUP;
S8:=CHREADY AND L<=CHDN;
ENTERLONG:DLY AND SPAN AND SID=8 AND B8 AND NOT(S8);
EXITLONG:DLY AND SPAN AND SID=8 AND S8;
