NDLY:=PERIOD=5;
NCOUNT:=BARSCOUNT(C);
NRNG:=H-L;
NIBS:=(C-L)/IF(NRNG>0,NRNG,1);
NDIF:=C-REF(C,1);
NUP:=SMA(MAX(NDIF,0),3,1);
NDN:=SMA(ABS(NDIF),3,1);
NRSI:=100*NUP/IF(NDN>0,NDN,1);
NEMA:=EMA(C,20);
NATR:=SMA(TR,20,1);
NB5:=MA(C,5)-1.5*STDP(C,5);
NB20:=MA(C,20)-2*STDP(C,20);
NMA20:=MA(C,20);
NMA50:=MA(C,50);
BC:NCOUNT,NODRAW;
B22:NDLY AND NCOUNT>=1 AND NRNG>0 AND NIBS<0.1,NODRAW;
S22:NDLY AND NCOUNT>=1 AND NRNG>0 AND NIBS>0.9,NODRAW;
B23:NDLY AND NCOUNT>=5 AND C<NB5,NODRAW;
S23:NDLY AND NCOUNT>=5 AND C>REF(H,1),NODRAW;
B24:NDLY AND NCOUNT>=4 AND NDN>0 AND NRNG>0 AND NRSI<30 AND NIBS<0.2,NODRAW;
S24:NDLY AND NCOUNT>=4 AND (C>REF(H,1) OR (NRNG>0 AND NIBS>0.9)),NODRAW;
B25:NDLY AND NCOUNT>=20 AND C<NEMA-2*NATR,NODRAW;
S25:NDLY AND NCOUNT>=20 AND C>NEMA,NODRAW;
B26:NDLY AND NCOUNT>=20 AND C<NB20,NODRAW;
S26:NDLY AND NCOUNT>=20 AND C>NMA20,NODRAW;
B27:NDLY AND NCOUNT>=51 AND CROSS(NMA20,NMA50),NODRAW;
S27:NDLY AND NCOUNT>=51 AND CROSS(NMA50,NMA20),NODRAW;
IBV:IF(NRNG>0,NIBS*1048576,-1),NODRAW;
RV3:IF(NDN>0,NRSI*1048576,-1),NODRAW;
KCM:NEMA*1048576,NODRAW;
KAT:NATR*1048576,NODRAW;
B5L:NB5*1048576,NODRAW;
B20L:NB20*1048576,NODRAW;
M20:NMA20*1048576,NODRAW;
M50:NMA50*1048576,NODRAW;
